    <script>

        const quizData = [

            {

                question: "Your engineering team tells you they are performing an 'archaeological dig' in the company repository. What are they actually doing?",

                options: [

                    "Searching for old bitcoin keys lost on an office hard drive in 2013.",

                    "Trying to figure out how a mission-critical piece of legacy COBOL or system code written decades ago works because the only person who understood it retired.",

                ],

                correct: 1,

                explain: "<strong>Technical Debt:</strong> Almost every major enterprise has a 'black box' legacy system that everyone is terrified to touch. Navigating it feels exactly like real archaeology."

            },

            {

                question: "What is the most common cause of sudden, unexplainable spikes in a company's monthly cloud computing bill?",

                options: [

                    "A massive, historic surge in organic global customer traffic.",

                    "A sophisticated, multi-layered cyberattack from a foreign state actor.",

                    "An intern or developer forgetting to turn off a massive, high-powered GPU testing cluster before going on vacation."

                ],

                correct: 2,

                explain: "<strong>Cloud FinOps:</strong> Unmonitored, runaway cloud instances left active by engineering environments are an incredibly common (and expensive) corporate pain point."

            },

            {

                question: "Your HR department is trying to hire a 'Prompt Engineer' for $250,000 a year. What is the core qualification listed on their resume?",

                options: [

                    "A PhD in computational physics and advanced linguistic architecture.",

                    "Knowing how to politely but firmly tell ChatGPT, 'Take a deep breath and think step-by-step,' to get it to output the correct spreadsheet format.",

                    "Being incredibly punctual to every single morning stand-up meeting."

                ],

                correct: 1,

                explain: "<strong>Prompt Engineering realities:</strong> While optimization takes skill, large language models respond shockingly well to human psychological cues, like asking them to 'think step-by-step' to eliminate logic errors."

            },

            {

                question: "Your cybersecurity team spent millions deploying multi-factor authentication and military-grade encryption. What remains the single greatest threat to your data security?",

                options: [

                    "A highly coordinated quantum-computing brute-force decryption attack.",

                    "Bob from Accounting clicking on an email that says, 'You won a free iPad, please type your corporate password here to claim it.'",

                    "Someone physically breaking into the office to steal a server rack."

                ],

                correct: 1,

                explain: "<strong>The Human Element:</strong> Corporate technology infrastructure is rarely the weakest link; social engineering and basic phishing remain the root cause of the vast majority of modern enterprise data breaches."

            },

            {

                question: "If a software development project running on an 'Agile methodology' is described as a 'Duck,' what does that mean?",

                options: [

                    "It looks perfectly smooth and calm on the surface, but is paddling furiously and chaotically beneath the water just to stay afloat.",

                    "It produces an incredible amount of loud noise but moves forward very slowly.",

                    "The architectural foundation migrates to a different cloud provider every winter."

                ],

                correct: 0,

                explain: "<strong>Project Delivery:</strong> The perfect metaphor for a sprint cycle where the external executive status dashboard report says 'Green/On Track,' while the underlying delivery team is working overtime to wire pieces together."

            }

        ];

 

        let currentQuestionIdx = 0;

        let score = 0;

 

        function initQuiz() {

            const container = document.getElementById('quiz-box-container');

            quizData.forEach((q, idx) => {

                const box = document.createElement('div');

                box.className = `question-box ${idx === 0 ? 'active' : ''}`;

                box.id = `q-${idx}`;

 

                let optionsHtml = '';

                q.options.forEach((opt, optIdx) => {

                    optionsHtml += `<li class="option-item" onclick="selectOption(${idx}, ${optIdx}, this)">${opt}</li>`;

                });

 

                box.innerHTML = `

                    <div class="question-num">Question ${idx + 1} of ${quizData.length}</div>

                    <div class="question-text">${q.question}</div>

                    <ul class="options-list">${optionsHtml}</ul>

                    <div class="explanation" id="explain-${idx}">${q.explain}</div>

                `;

                container.appendChild(box);

            });

            updateProgressBar();

        }

 

        function updateProgressBar() {

            const progress = document.getElementById('progress');

            const percent = ((currentQuestionIdx) / quizData.length) * 100;

            progress.style.width = `${percent}%`;

        }

 

        function selectOption(qIdx, optIdx, element) {

            const q = quizData[qIdx];

            const parent = element.parentElement;

            const items = parent.getElementsByClassName('option-item');

 

            for (let item of items) {

                item.classList.add('disabled');

            }

 

            if (optIdx === q.correct) {

                element.classList.add('correct');

                score++;

            } else {

                element.classList.add('incorrect');

                items[q.correct].classList.add('correct');

            }

 

            document.getElementById(`explain-${qIdx}`).style.display = 'block';

            document.getElementById('next-btn').style.display = 'inline-block';

        }

 

        function nextQuestion() {

            document.getElementById('next-btn').style.display = 'none';

            document.getElementById(`q-${currentQuestionIdx}`).classList.remove('active');

           

            currentQuestionIdx++;

            updateProgressBar();

 

            if (currentQuestionIdx < quizData.length) {

                document.getElementById(`q-${currentQuestionIdx}`).classList.add('active');

            } else {

                showResults();

            }

        }

 

        function showResults() {

            document.getElementById('progress').style.width = '100%';

            document.getElementById('quiz-box-container').style.display = 'none';

           

            const resultsScreen = document.getElementById('results-screen');

            resultsScreen.style.display = 'block';

           

            document.getElementById('score-text').innerText = `${score} / ${quizData.length}`;

           

            let verdict = "";

            if (score === 5) verdict = "👑 Master of Corporate Reality: You have a flawless understanding of engineering culture, tech debt, and how things actually work.";

            else if (score >= 3) verdict = "🚀 Seasoned Tech Leader: You recognize the real-world organizational hurdles behind engineering buzzwords.";

            else verdict = "📈 Optimistic Visionary: You trust the PowerPoint presentations implicitly. A good time to buy your head of infrastructure a coffee!";

           

            document.getElementById('verdict-text').innerText = verdict;

        }

 

        window.onload = initQuiz;

    </script>
